# v1.8.0 Single & Dual Link Preview Pipeline Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement a mode-aware preview pipeline that branches between Dual Link (Chip 0 master + MCU UART in-band tunnel with BlueRiver fallback) and Single Link (direct Semtech BlueRiver Stream 1 Scaler node), enhances `UnifiedRtpDemuxReceiver` to listen concurrently on dual ports (5000 and 5004) with socket reuse and source-IP whitelisting, enforces deterministic hardware teardown via `address 0.0.0.0` and `auto_stop: true`, and introduces a clean chip preview toggle (`[ Chip 0 | Chip 1 ]`) on Single Link preview cards.

**Architecture:** 
- **Control Plane**: In `SdvoeClient`, mode-aware stream dispatch sends MCU UART commands for `LinkMode == "DUAL"`, while directly issuing `set <MAC> stream HDMI 1 enable true address <MCAST_IP>` (Stream 1 Scaler node) and `auto_stop: true` for `LinkMode == "SINGLE"`. Teardown explicitly issues `set <MAC> stream HDMI <idx> enable false address 0.0.0.0` to command the AVP packetizer FIFO to halt.
- **Media Plane**: In `UnifiedRtpDemuxReceiver`, dual `UdpClient` sockets bind concurrently to port 5000 and port 5004 with `ReuseAddress = true` and 8MB buffers, feeding a unified RFC 4175 demultiplexing loop with strict source-IP filtering.
- **Presentation Layer**: In `EncoderCardViewModel` and `EncoderCardView.xaml`, Single Link cards with a valid companion chip expose a segmented `[ Chip 0 | Chip 1 ]` selector that safely tears down and re-seeds the stream on the selected chip without leaving orphan multicast subscriptions.

**Tech Stack:** C# 12, .NET 8.0, WPF, Semtech BlueRiver CLI/Telnet & REST JSON-RPC, UDP Multicast Sockets (RFC 4175), XUnit.

**Spec:** `docs/superpowers/specs/2026-10-24-v1.8.0-single-and-dual-link-preview-pipeline.md`

## Global Constraints

- Preserve all existing comments and docstrings.
- Version bump: Increment assembly and package version from `1.7.0` to `1.8.0`.
- Maintain 100% test pass rate across the test suite (baseline: 565 tests).
- All socket bindings must enforce `ExclusiveAddressUse = false` and `SocketOptionName.ReuseAddress = true`.
- Zero cross-thread WinForms/WPF UI mutations from background socket threads.

## Review Focus

1. **Unregistered Source IP Flood**: Packets arriving from rogue network nodes on ports 5000 or 5004 must be dropped immediately by `TryResolveChannel` before allocating any buffer or header structures.
2. **Rapid Stream Re-route / Chip Switch**: Toggling between Chip 0 and Chip 1 on a Single Link card must issue `enable false address 0.0.0.0` to the old chip before commanding the new chip, preventing desynchronized RTP timestamps.
3. **Dual Link Fallback**: If an AVAS device in Dual Link mode does not respond to MCU UART within the timeout window, the client must seamlessly fall back to BlueRiver stream start.
4. **NIC Multi-Adapter Multicast Join**: When joining multicast groups on dual sockets, the receiver must join on the designated local NIC adapter first, falling back to multi-adapter sweep without throwing uncaught socket exceptions.
5. **Backwards Compatibility**: Existing `StartPreviewStreamAsync` and `StopPreviewStreamAsync` overloads must remain binary-compatible and function transparently for callers passing default parameters.

public async Task<bool> StartSingleLinkStreamAsync(string macAddress, string multicastIp, int port = 5004, int streamIndex = 1, CancellationToken ct = default)
{
    string normMac = NormalizeMac(macAddress);
    AppLogger.Info("SdvoeClient", $"Starting Single-Link native BlueRiver Stream {streamIndex} (Scaler) for {normMac} -> {multicastIp}:{port}");

    bool telnetOk = false;
    try
    {
        if (!_isTelnetAuthenticated) await ConnectTelnetAsync(ct);
        if (IsTelnetConnected)
        {
            string cmdEnable = $"set {normMac} stream HDMI {streamIndex} enable true address {multicastIp}";
            string resp1 = await SendTelnetCommandAsync(cmdEnable, ct);

            string cmdAutoStop = $"set {normMac} stream HDMI {streamIndex} auto_stop true";
            string resp2 = await SendTelnetCommandAsync(cmdAutoStop, ct);

            telnetOk = IsCommandSuccessful(resp1) || IsRestResponseSuccess(resp1);
        }
    }
    catch (Exception ex)
    {
        AppLogger.Warn("SdvoeClient", $"Telnet StartSingleLinkStreamAsync failed on {normMac}: {ex.Message}");
    }

    if (telnetOk) return true;

    // REST fallback
    try
    {
        string path = $"/api/device/{normMac}";
        var payload = new
        {
            op = "set",
            streams = new[]
            {
                new
                {
                    type = "HDMI",
                    index = streamIndex,
                    configuration = new
                    {
                        enable = true,
                        address = multicastIp,
                        auto_stop = true
                    }
                }
            }
        };
        using var resp = await PostDeviceRestAsync(path, payload, ct);
        if (resp != null && resp.IsSuccessStatusCode)
        {
            string body = await resp.Content.ReadAsStringAsync(ct);
            return IsRestResponseSuccess(body) || IsCommandSuccessful(body);
        }
    }
    catch (Exception ex)
    {
        AppLogger.Error("SdvoeClient", $"REST StartSingleLinkStreamAsync failed on {normMac}", ex);
    }

    return false;
}

public async Task<bool> StartPreviewStreamAsync(
    string macAddress,
    string multicastIp,
    int port,
    string linkMode = "UNKNOWN",
    int chipIndex = 0,
    CancellationToken ct = default)
{
    string normMac = NormalizeMac(macAddress);

    // If device is explicitly in SINGLE link mode, directly invoke native BlueRiver Stream 1 Scaler
    if (string.Equals(linkMode, "SINGLE", StringComparison.OrdinalIgnoreCase))
    {
        if (await StartSingleLinkStreamAsync(normMac, multicastIp, port, streamIndex: 1, ct))
        {
            return true;
        }
    }

    // Otherwise (DUAL link mode or UNKNOWN), prioritize AVAS-223 native AVP RS-232 tunneling
    if (await StartPreviewStreamViaAvpRs232Async(normMac, multicastIp, ct))
    {
        AppLogger.Info("SdvoeClient", $"Started AVAS-223 AVP thumbnail stream via RS-232 for {normMac} -> {multicastIp}");
        return true;
    }

    // Fall back to standard Semtech BlueRiver thumbnail stream commands
    await ConfigureThumbnailStreamAsync(normMac, 1.0, 12345, port, ct);

    string cmd = $"start {normMac}:thumbnail:0 {multicastIp}";
    try
    {
        if (!_isTelnetAuthenticated) await ConnectTelnetAsync(ct);
        string resp = await SendTelnetCommandAsync(cmd, ct);
        if (IsCommandSuccessful(resp)) return true;
    }
    catch { }

    if (await StartPreviewStreamViaRestAsync(normMac, multicastIp, ct))
    {
        return true;
    }

    // Final fallback to native BlueRiver Stream 1 Scaler
    return await StartSingleLinkStreamAsync(normMac, multicastIp, port, streamIndex: 1, ct);
}

        public void EnsureStarted(IEnumerable<int>? ports = null)
        {
            lock (_socketLock)
            {
                if (_isListening) return;

                var targetPorts = (ports != null && ports.Any()) ? ports.Distinct().ToArray() : DefaultPorts;
                _cts = new CancellationTokenSource();
                _isListening = true;
                _udpClients.Clear();
                _listeningPorts.Clear();

                try
                {
                    foreach (int p in targetPorts)
                    {
                        try
                        {
                            var client = new UdpClient();
                            client.ExclusiveAddressUse = false;
                            client.Client.SetSocketOption(SocketOptionLevel.Socket, SocketOptionName.ReuseAddress, true);
                            client.Client.ReceiveBufferSize = 8 * 1024 * 1024; // 8MB buffer

                            var localEp = new IPEndPoint(IPAddress.Any, p);
                            client.Client.Bind(localEp);

                            _udpClients.Add(client);
                            _listeningPorts.Add(p);

                            var token = _cts.Token;
                            Task.Run(() => ReceiveLoopAsync(client, token), token);
                            AppLogger.Info("UnifiedDemux", $"Unified RTP Demux listener active on port {p}");
                        }
                        catch (Exception ex)
                        {
                            AppLogger.Warn("UnifiedDemux", $"Failed to bind UDP port {p}: {ex.Message}");
                        }
                    }

                    if (_udpClients.Count == 0)
                    {
                        throw new InvalidOperationException("Could not bind any UDP listening ports.");
                    }

                    _port = _listeningPorts[0];

                    // Re-join any previously tracked multicast groups across all clients
                    foreach (var mcastIp in _joinedMulticastGroups)
                    {
                        _groupLocalNics.TryGetValue(mcastIp, out var localNic);
                        JoinMulticastGroupInternal(mcastIp, localNic);
                    }
                }
                catch (Exception ex)
                {
                    AppLogger.Error("UnifiedDemux", "Failed to initialize unified RTP sockets", ex);
                    StopListeningInternal();
                    throw;
                }
            }
        }

        public void WireUpCardChipSelection(EncoderCardViewModel card)
        {
            card.OnChipSelectionChanged = async (c, targetChip) =>
            {
                string oldMac = targetChip == 1 ? c.MacAddress : c.CompanionMac;
                string newMac = targetChip == 1 ? c.CompanionMac : c.MacAddress;

                AppLogger.Info("StreamControl", $"Switching preview stream for card {c.DeviceName} from {oldMac} (Chip {1 - targetChip}) to {newMac} (Chip {targetChip})");

                c.StatusMessage = $"Switching to Chip {targetChip}...";

                // Teardown previous stream
                await _multicastController.StopPreviewStreamAsync(oldMac, c.LinkMode, 1 - targetChip, CancellationToken.None);
                await Task.Delay(100);

                // Start new stream on the target chip
                bool started = await _multicastController.StartPreviewStreamAsync(newMac, c.MulticastIp, c.Port, c.LinkMode, targetChip, CancellationToken.None);
                c.IsStreaming = started;
                c.StatusMessage = started ? $"Streaming (Chip {targetChip})" : "Stream restart failed";

                // Update channel alias and re-attach
                var aliases = new List<string> { c.MacAddress };
                if (!string.IsNullOrEmpty(c.CompanionMac)) aliases.Add(c.CompanionMac);

                var channel = UnifiedRtpDemuxReceiver.Instance.GetOrCreateChannel(c.UnicastIp, c.MulticastIp, aliases);
                c.AttachReceiver(channel);
            };
        }
